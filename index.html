Desenvolvimento Full-Stack: Controle de Máquinas e Manutenção

Visão Geral do Projeto

Neste treinamento prático, vamos construir um sistema industrial completo. O projeto é dividido em duas grandes camadas que conversam entre si:

  • O Back-end (A "Mente"): Feito em Python com FastAPI. É o servidor que processa as regras de negócio, valida dados de técnicos e salva informações no banco de dados SQLite usando SQLAlchemy.
  • O Front-end (O "Rosto"): Feito em HTML, CSS e JavaScript. É o painel visual onde o operador clica, digita e visualiza os relatórios. Ele se comunica com o Back-end através do comando fetch.

Passo 2: Conexão com o Banco (database.py)

Configure o motor do SQLAlchemy para apontar para um arquivo SQLite local (maquinas.db).

from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker, declarative_base

SQLALCHEMY_DATABASE_URL = "sqlite:///./maquinas.db"

engine = create_engine(
    SQLALCHEMY_DATABASE_URL, connect_args={"check_same_thread": False}
)
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)
Base = declarative_base()

def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

Passo 3: As Tabelas e Relacionamentos (models.py)

Defina as tabelas de Usuários, Máquinas e Manutenções. A tabela de manutenções conecta o técnico ao equipamento usando chaves estrangeiras.

from sqlalchemy import Column, Integer, String, Boolean, ForeignKey, DateTime
from sqlalchemy.orm import relationship
from datetime import datetime
from database import Base

class Usuario(Base):
    __tablename__ = "usuarios"
    id = Column(Integer, primary_key=True, index=True)
    nome = Column(String, nullable=False)
    email = Column(String, unique=True, index=True, nullable=False)
    manutencoes = relationship("Manutencao", back_populates="usuario")

class Maquina(Base):
    __tablename__ = "maquinas"
    id = Column(Integer, primary_key=True, index=True)
    nome = Column(String, nullable=False)
    marca = Column(String, nullable=False)
    numero_serie = Column(String, unique=True, index=True, nullable=False)
    em_operacao = Column(Boolean, default=True)
    manutencoes = relationship("Manutencao", back_populates="maquina")

class Manutencao(Base):
    __tablename__ = "manutencoes"
    id = Column(Integer, primary_key=True, index=True)
    descricao = Column(String, nullable=False)
    data = Column(DateTime, default=datetime.utcnow)
    maquina_id = Column(Integer, ForeignKey("maquinas.id"), nullable=False)
    usuario_id = Column(Integer, ForeignKey("usuarios.id"), nullable=False)

    maquina = relationship("Maquina", back_populates="manutencoes")
    usuario = relationship("Usuario", back_populates="manutencoes")

Passo 4: Validação de Dados (schemas.py)

Use o Pydantic para garantir que a API só aceite dados corretos.

from pydantic import BaseModel
from typing import Optional
from datetime import datetime

class UsuarioBase(BaseModel):
    nome: str
    email: str
class UsuarioCreate(UsuarioBase): pass
class UsuarioResponse(UsuarioBase):
    id: int
    class Config: from_attributes = True

class MaquinaBase(BaseModel):
    nome: str
    marca: str
    numero_serie: str
    em_operacao: bool = True
class MaquinaCreate(MaquinaBase): pass
class MaquinaResponse(MaquinaBase):
    id: int
    class Config: from_attributes = True

class ManutencaoBase(BaseModel):
    descricao: str
    maquina_id: int
    usuario_id: int
class ManutencaoCreate(ManutencaoBase): pass
class ManutencaoResponse(ManutencaoBase):
    id: int
    data: datetime
    maquina: Optional[MaquinaResponse] = None
    usuario: Optional[UsuarioResponse] = None
    class Config: from_attributes = True

Passo 6: A API e a Liberação do CORS (main.py)

Aqui criamos as rotas e ativamos o CORSMiddleware. Sem o CORS, o navegador bloqueará o Front-end de conversar com o Back-end por medidas de segurança.

from fastapi import FastAPI, Depends, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from sqlalchemy.orm import Session
from typing import List
import models, schemas
from database import get_db

app = FastAPI(title="API de Manutenção")

# Libera o acesso do Front-end (CORS)
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"], 
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

@app.post("/usuarios/", response_model=schemas.UsuarioResponse)
def criar_usuario(usuario: schemas.UsuarioCreate, db: Session = Depends(get_db)):
    if db.query(models.Usuario).filter(models.Usuario.email == usuario.email).first():
        raise HTTPException(status_code=400, detail="Email já existe.")
    novo = models.Usuario(**usuario.model_dump())
    db.add(novo)
    db.commit()
    db.refresh(novo)
    return novo

@app.get("/usuarios/", response_model=List[schemas.UsuarioResponse])
def listar_usuarios(db: Session = Depends(get_db)):
    return db.query(models.Usuario).all()

@app.post("/maquinas/", response_model=schemas.MaquinaResponse)
def criar_maquina(maquina: schemas.MaquinaCreate, db: Session = Depends(get_db)):
    if db.query(models.Maquina).filter(models.Maquina.numero_serie == maquina.numero_serie).first():
        raise HTTPException(status_code=400, detail="Série já cadastrada.")
    nova = models.Maquina(**maquina.model_dump())
    db.add(nova)
    db.commit()
    db.refresh(nova)
    return nova

@app.get("/maquinas/", response_model=List[schemas.MaquinaResponse])
def listar_maquinas(db: Session = Depends(get_db)):
    return db.query(models.Maquina).all()

@app.post("/manutencoes/", response_model=schemas.ManutencaoResponse)
def registrar_manutencao(manu: schemas.ManutencaoCreate, db: Session = Depends(get_db)):
    if not db.query(models.Maquina).filter(models.Maquina.id == manu.maquina_id).first():
        raise HTTPException(status_code=404, detail="Máquina não encontrada.")
    if not db.query(models.Usuario).filter(models.Usuario.id == manu.usuario_id).first():
        raise HTTPException(status_code=404, detail="Usuário não encontrado.")

    nova_manu = models.Manutencao(**manu.model_dump())
    db.add(nova_manu)
    
    # Atualiza o status da máquina
    db.query(models.Maquina).filter(models.Maquina.id == manu.maquina_id).update({"em_operacao": True})
    
    db.commit()
    db.refresh(nova_manu)
    return nova_manu

@app.get("/manutencoes/", response_model=List[schemas.ManutencaoResponse])
def listar_manutencoes(db: Session = Depends(get_db)):
    return db.query(models.Manutencao).all()
PARTE 2: O FRONT-END (O PAINEL VISUAL)

Passo 7: Consumindo a API com JavaScript

Crie um arquivo chamado dashboard.html. O Front-end usa a função fetch() para enviar dados (POST) ou buscar dados (GET) do FastAPI.

Abaixo está um resumo de como o HTML e o JavaScript operam juntos. Quando o usuário clica em "Cadastrar Técnico", o script pega os valores dos inputs e envia para http://127.0.0.1:8000/usuarios/.

<!-- Exemplo da Estrutura Visual -->
<h2>Cadastrar Técnico</h2>
<form id="formUsuario">
    <input type="text" id="nomeUsuario" placeholder="Nome" required>
    <input type="email" id="emailUsuario" placeholder="Email" required>
    <button type="submit">Cadastrar</button>
</form>

<script>
    const API_URL = "http://127.0.0.1:8000";

    // Enviando dados para o Back-end (POST)
    document.getElementById('formUsuario').addEventListener('submit', async (e) => {
        e.preventDefault();
        
        // Empacota os dados
        const payload = {
            nome: document.getElementById('nomeUsuario').value,
            email: document.getElementById('emailUsuario').value
        };
        
        // Dispara para o servidor
        const res = await fetch(`${API_URL}/usuarios/`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload)
        });
        
        if (res.ok) alert('Técnico salvo com sucesso no banco!');
    });

    // Buscando dados do Back-end (GET)
    async function carregarMaquinas() {
        const res = await fetch(`${API_URL}/maquinas/`);
        const maquinas = await res.json();
        console.log(maquinas); // Mostra a lista vinda do banco
    }
</script>

Passo 8: Rodando o Sistema Completo

Para ver a mágica acontecer, as duas partes precisam rodar simultaneamente:

  1. Ligue o Motor (Servidor): No terminal do VS Code, inicie a API rodando:
    uvicorn main:app --reload
  2. Abra a Interface: Dê um duplo clique no arquivo dashboard.html (ele abrirá no seu navegador Chrome/Edge).
  3. Teste o Fluxo: Cadastre um técnico pelo painel HTML, depois cadastre uma máquina, e por fim gere uma Ordem de Serviço vinculando os dois. Tudo será validado pelo Python e salvo no SQLite.
